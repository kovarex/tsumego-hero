export interface LineChartSeries
{
	label: string;
	color: string;
	//the left or the right value axis, each with its own scale
	axis: 'left' | 'right';
	//whether the area under the line is filled
	fill: boolean;
	//[date, value], the date as YYYY-MM-DD; null leaves a gap in the line
	points: [string, number | null][];
}
 
interface Props
{
	title: string | null;
	series: LineChartSeries[];
	height: number;
	//what one point stands for, it names the date under the mouse
	period: 'month' | 'week' | 'day';
	//false fits the value axes to the values, for a rating that never gets near zero
	startAtZero: boolean;
	//under the title, as the ranges of the rating chart
	header?: ReactNode;
	//the left axis of a rating in go ranks: its ticks and grid lines at the lower edges of the ranks, named like "5k"
	//in the chart's text colour rather than the colour of the series, and the tooltip gives the rating as a plain
	//number with its rank, "2875 (12d)"
	rankAxis?: boolean;
	//the date of a point still going on, which the tooltip names "Today so far"
	partialDate?: string;
}
 
interface Point
{
	date: string;
	at: ChartDate;
	value: number;
}
 
const MARGIN_TOP = 10;
const MARGIN_BOTTOM = 26;
const AXIS_GAP = 8;
const CHARACTER_WIDTH = 7;
const TICK_SPACING_X = 70;
const TICK_SPACING_Y = 45;
const AREA_OPACITY = 0.25;
const TOOLTIP_OFFSET = 12;
 
//The runs of points of one series without a gap, each drawn as one stretch of the line.
function stretchesOf(points: [string, number | null][]): Point[][]
{
	const stretches: Point[][] = [[]];
	for (const [date, value] of points)
		if (value === null)
			stretches.push([]);
		else
			stretches[stretches.length - 1].push({ date, at: parseChartDate(date), value });
	return stretches.filter(stretch => stretch.length > 0);
}
