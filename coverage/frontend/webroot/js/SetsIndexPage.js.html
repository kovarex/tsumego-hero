	let position = -1;

		for(let i=0;i<allTopicTiles.length;i++)

function applyNearestRangeEnd(index){

	if(from > to){
		let swap = from;

		to = swap;
	}
	if(index <= from)
		applyRankRange(index, to);
	else if(index >= to)
		applyRankRange(from, index);

		applyRankRange(index, to);

}
